import os
from dotenv import load_dotenv
from web3 import Web3

# Carregar variáveis de ambiente
load_dotenv()

# Configurações seguras
PRIVATE_KEY = os.getenv("a9947d3ce29c271d055cfc67dd579b5c6306f8a458fb679f9c2ada57a6fd99bf")
INFURA_URL = os.getenv("https://sepolia.infura.io/v3/fc82183aadf3423d8c59c98d97334930")
WALLET_ADDRESS = os.getenv("0xAA2472F217bAc2d3d8231d5206Ac0C94fED51403")

# Conectar à rede Ethereum (Sepolia neste caso)
web3 = Web3(Web3.HTTPProvider(INFURA_URL))

if web3.isConnected():
    print("Conectado à rede de testes Ethereum!")
else:
    print("Falha na conexão com a rede.")

# Exemplo de uso seguro da chave privada
def enviar_transacao(destinatario, valor):
    try:
        # Preparar a transação
        transacao = {
            'to': destinatario,
            'value': web3.toWei(valor, 'ether'),
            'gas': 21000,
            'gasPrice': web3.toWei('20', 'gwei'),
            'nonce': web3.eth.getTransactionCount(WALLET_ADDRESS),
            'chainId': 11155111  # ID da rede Sepolia
        }

        # Assinar a transação
        assinado = web3.eth.account.sign_transaction(transacao, PRIVATE_KEY)

        # Enviar a transação
        tx_hash = web3.eth.send_raw_transaction(assinado.rawTransaction)
        print(f"Transação enviada com sucesso! Hash: {web3.toHex(tx_hash)}")
    except Exception as e:
        print(f"Erro ao enviar a transação: {str(e)}")

# Exemplo de chamada
enviar_transacao("0x3068357b654aa87750420dD9a95469713a133e93", 0.01)
