def ajustar_dataframe(df):
    """
    Ajusta o DataFrame para garantir que ele tenha as mesmas colunas usadas no treinamento do modelo.
    """
    # Definir as colunas que o modelo espera
    colunas_modelo = ['SMA_10', 'SMA_50', 'close', 'volume']
    
    # Verifique se todas as colunas necessárias estão no DataFrame
    missing_cols = [col for col in colunas_modelo if col not in df.columns]
    for col in missing_cols:
        df[col] = 0  # Preencher as colunas ausentes com 0 ou outro valor padrão
    
    # Certificar-se de que o DataFrame tem apenas as colunas necessárias
    df = df[colunas_modelo]
    
    return df
